import { expect, test } from "@playwright/test";

import { CartPage } from "../../pages/CartPage.js";
import { PaymentPage } from "../../pages/PaymentPage.js";
import { ConfirmationPage } from "../../pages/ConfirmationPage.js";
import { loginFlow } from "../../flows/login.flow.js";
import { clearCartFlow } from "../../flows/clear-cart.flow.js";
import { couponCode, cardNumber } from "../../fixtures/environment.js";

/** TC-77 — generated by Specra from the Test Model. */
test("Checkout a cart of two items @regression @checkout", async ({ page }) => {
  // Exercises every action group: a reused flow, waits, input variants, a drag, and the selector escape hatch.
  const cartPage = new CartPage(page);
  const paymentPage = new PaymentPage(page);
  const confirmationPage = new ConfirmationPage(page);

  // Setup
  // Sign in before the case begins
  await loginFlow(page);

  await cartPage.goto();
  await expect(cartPage.lineItems).toHaveCount(2);
  await cartPage.shippingMethod.selectOption("express");
  await cartPage.giftWrap.check();
  await cartPage.couponInput.fill(couponCode());
  await cartPage.couponInput.press("Enter");
  // The legacy promo widget has no stable handle yet
  await page.locator("#promo .handle").dragTo(cartPage.promoDropZone);
  await cartPage.checkoutButton.click();
  await paymentPage.cardNumberInput.fill(cardNumber());
  await paymentPage.payButton.click();
  await expect(page).toHaveURL(new RegExp("^/orders/[0-9]+$"));
  await expect(confirmationPage.status).toContainText("Thank you");
  await expect(confirmationPage.downloadInvoice).toHaveAttribute("aria-disabled", "false");

  // Teardown
  await clearCartFlow(page);
});
